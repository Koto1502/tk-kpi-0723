How ad tickets are granted

87% of ad tickets outside China are granted with no ad activity beforehand.

For each ads_lucky_machine_ticket grant, the same user's ad events in the preceding 60 seconds are classified: an impression on the event placement (expected path), an impression elsewhere, an ad show or reward without impression, a failed load, a tap only, or nothing. Grants are issued in bursts roughly one second apart, approximately the full daily allowance at once, which is not compatible with one rewarded ad per ticket (a rewarded ad lasts 15–30 seconds).

WITH x AS (
  SELECT user_pseudo_id u, event_name ev, event_timestamp ts, geo.country = 'China' cn, app_info.version v,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='position') pos,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='ad_type') ad_type,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='status') status,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='error') err,
    COALESCE((SELECT value.string_value FROM UNNEST(event_params) WHERE key='ad_placement'),
             (SELECT value.string_value FROM UNNEST(event_params) WHERE key='placement')) placement
  FROM `tiny-knightfall-idle-rpg-game.analytics_538412813.events_intraday_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260916' AND '20260925'
    AND event_name IN ('earn', 'show_ad_click', 'ad_request', 'ad_request_status', 'ad_show',
                       'ad_impression_MAX', 'ad_reward', 'ad_rewarded_show_failed', 'ad_rewarded_rewareded_mini_event')
),
grants AS (SELECT u, ts, cn, v FROM x WHERE ev = 'earn' AND pos = 'ads_lucky_machine_ticket'),
ctx AS (
  SELECT g.u, g.ts, g.cn, g.v,
    COUNTIF(a.ev = 'show_ad_click')                                         taps,
    COUNTIF(a.ev = 'ad_request_status' AND a.ad_type = 'rewarded' AND a.status = 'success') rewarded_loaded,
    COUNTIF(a.ev = 'ad_request_status' AND a.status = 'fail')               load_failures,
    COUNTIF(a.ev = 'ad_show')                                               ad_shows,
    COUNTIF(a.ev = 'ad_impression_MAX')                                     impressions,
    COUNTIF(a.ev = 'ad_impression_MAX' AND a.placement = 'rewareded_mini_event') event_impressions,
    COUNTIF(a.ev = 'ad_reward')                                             ad_rewards,
    COUNTIF(a.ev = 'ad_rewarded_rewareded_mini_event')                      mini_event_ad_logs
  FROM grants g LEFT JOIN x a
    ON a.u = g.u AND a.ev != 'earn' AND a.ts BETWEEN g.ts - 60000000 AND g.ts
  GROUP BY 1, 2, 3, 4
)
SELECT IF(cn, 'China', 'rest of world') region,
  CASE
    WHEN event_impressions > 0 THEN '1 event ad shown (impression on rewareded_mini_event)'
    WHEN impressions > 0       THEN '2 an ad was shown, but on another placement'
    WHEN ad_shows > 0 OR ad_rewards > 0 THEN '3 ad show/reward logged, no impression'
    WHEN load_failures > 0     THEN '4 no ad: load failed, ticket granted anyway'
    WHEN taps > 0              THEN '5 no ad: watch-ad tap only'
    ELSE                            '6 no ad activity at all in the preceding 60 s'
  END AS what_preceded_the_grant,
  COUNT(*) grants, COUNT(DISTINCT u) users
FROM ctx GROUP BY 1, 2 ORDER BY 1, 2

-- Events logged on v29 but missing (or nearly missing) on v32-v34, 2026-09-18 .. 2026-10-01, ex-China.
-- Per-user rates so builds with different audience sizes compare fairly.
WITH e AS (
  SELECT event_name, app_info.version v, user_pseudo_id u
  FROM `tiny-knightfall-idle-rpg-game.analytics_538412813.events_intraday_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260918' AND '20261001' AND geo.country != 'China'
    AND app_info.version IN ('29', '32', '33', '34')
),
users AS (SELECT v, COUNT(DISTINCT u) n FROM e GROUP BY v),
ev AS (SELECT event_name, v, COUNT(*) events, COUNT(DISTINCT u) users FROM e GROUP BY 1, 2)
SELECT event_name,
  MAX(IF(ev.v='29', ev.users, 0)) users_v29, MAX(IF(ev.v='32', ev.users, 0)) users_v32,
  MAX(IF(ev.v='33', ev.users, 0)) users_v33, MAX(IF(ev.v='34', ev.users, 0)) users_v34,
  ROUND(100 * MAX(IF(ev.v='29', ev.users / users.n, 0)), 1) pct_users_v29,
  ROUND(100 * MAX(IF(ev.v='34', ev.users / users.n, 0)), 1) pct_users_v34
FROM ev JOIN users USING (v)
GROUP BY event_name
HAVING users_v29 >= 20 AND pct_users_v34 < 0.25 * pct_users_v29
ORDER BY users_v29 DESC
