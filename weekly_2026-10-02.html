-- China vs rest, 2026-09-18 .. 2026-10-01, by app version: who they are and what they earn
WITH e AS (
  SELECT user_pseudo_id u, app_info.version v, geo.country = 'China' AS china, event_name ev, event_value_in_usd usd,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='ad_type') ad_type,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='status')  status,
    (SELECT COALESCE(value.double_value, value.float_value, CAST(value.int_value AS FLOAT64))
       FROM UNNEST(event_params) WHERE key='value') val,
    (SELECT value.int_value FROM UNNEST(event_params) WHERE key='engagement_time_msec') eng
  FROM `tiny-knightfall-idle-rpg-game.analytics_538412813.events_intraday_*`
  WHERE _TABLE_SUFFIX BETWEEN '20260918' AND '20261001'
)
SELECT IF(china, 'China', 'rest of world') AS region, v AS version,
  COUNT(DISTINCT u)                                                       users,
  COUNT(DISTINCT IF(ev='first_open', u, NULL))                            installs,
  ROUND(SUM(IF(ev='user_engagement', eng, 0)) / 60000 / COUNT(DISTINCT u), 1) minutes_per_user,
  COUNTIF(ev='purchase_click')                                            shop_taps,
  COUNTIF(ev='in_app_purchase')                                           purchases,
  ROUND(SUM(IF(ev='in_app_purchase', usd, 0)), 2)                         iap_usd,
  COUNTIF(ev='ad_impression_MAX')                                         ad_impressions,
  ROUND(SUM(IF(ev='ad_impression_MAX', val, 0)), 2)                       ad_usd,
  ROUND(100 * SAFE_DIVIDE(COUNTIF(ev='ad_request_status' AND ad_type='rewarded' AND status='success'),
                          COUNTIF(ev='ad_request_status' AND ad_type='rewarded')), 1) rewarded_fill_pct
FROM e
GROUP BY region, version
HAVING users >= 30
ORDER BY region, users DESC
