-- Events logged on v29 but missing (or nearly missing) on v32-v34, 2026-09-18 .. 2026-10-01, ex-China.

WITH e AS (
  SELECT event_name, app_info.version v, user_pseudo_id u
  FROM `tiny-knightfall-idle-rpg-game.analytics_538412813.events_intraday_*`

),

ev AS (SELECT event_name, v, COUNT(*) events, COUNT(DISTINCT u) users FROM e GROUP BY 1, 2)
SELECT event_name,
  MAX(IF(ev.v='29', ev.users, 0)) users_v29, MAX(IF(ev.v='32', ev.users, 0)) users_v32,
